cmake_minimum_required(VERSION 3.20)
project(lab3_market_forecast LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED)
add_executable(forecast_omp src/forecast_omp.cpp)
target_link_libraries(forecast_omp PRIVATE OpenMP::OpenMP_CXX)
if(MSVC)
  target_compile_options(forecast_omp PRIVATE /O2)
else()
  target_compile_options(forecast_omp PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
